22: /// Returns whether it is possible to pack lhs and rhs into a unique
23: /// ciphertext without exceeding the max storable value using the formula:
24: /// `unique_ciphertext = (lhs * factor) + rhs`
25: fn ciphertexts_can_be_packed_without_exceeding_space_or_noise(

27:     lhs: CiphertextNoiseDegree,
28:     rhs: CiphertextNoiseDegree,
29:     factor: usize,
30: ) -> Result<(), CheckError> {
31:     let final_degree = (lhs.degree * factor) + rhs.degree;
32: 
33:     // Do not use server key max degree as it may be smaller (to keep carry propagation margin)
34:     let max_degree =
35:         MaxDegree::from_msg_carry_modulus(server_key.message_modulus, server_key.carry_modulus);
36: 
37:     max_degree.validate(final_degree)?;
38: 
39:     let final_noise_level = (lhs.noise_level * factor) + rhs.noise_level;
40: 
41:     server_key.max_noise_level.validate(final_noise_level)?;

43:     if rhs.degree.get() >= factor {
44:         return Err(CheckError::UnscaledScaledOverlap {
45:             unscaled_degree: rhs.degree,
46:             scale: factor as u8,
47:         });
48:     }

50:     Ok(())
51: }
