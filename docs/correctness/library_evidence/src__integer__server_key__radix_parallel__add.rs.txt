44:     // However that penalty only kicks in when certain level of
45:     // parallelism is used
46:     let penalty_threshold = num_threads / 2;

48:     // The unit of latency is a PBS
49:     let compute_latency_of_one_layer = |num_blocks: usize, num_threads: usize| -> usize {
50:         let latency = num_blocks.div_ceil(num_threads);
51:         if num_blocks >= penalty_threshold {
52:             latency * PARALLEL_LATENCY_PENALTY
53:         } else {
54:             latency
55:         }
56:     };

58:     // Estimate the latency of the parallelized algorithm
59:     // One pre-processing layer, one layer to compute what happens in each grouping,
60:     // one final post processing layer
61:     let mut parallel_expected_latency = 3 * compute_latency_of_one_layer(num_blocks, num_threads);

63:     let grouping_size = full_modulus.ilog2();
64:     let num_groups = num_blocks.div_ceil(grouping_size as usize);
65: 
66:     let num_carry_to_resolve = num_groups.saturating_sub(1);
67: 
68:     let sequential_depth = (num_carry_to_resolve.saturating_sub(1) as u32) / (grouping_size - 1);
69:     let hillis_steel_depth = if num_carry_to_resolve == 0 {
70:         0
71:     } else {
72:         num_carry_to_resolve.ceil_ilog2()
73:     };
74: 
75:     let parallel_algo_uses_sequential_to_resolve_grouping_carries =
76:         sequential_depth <= hillis_steel_depth;
77: 
78:     if parallel_algo_uses_sequential_to_resolve_grouping_carries {
79:         parallel_expected_latency += sequential_depth as usize
80:             * compute_latency_of_one_layer(grouping_size as usize, num_threads);
81:     } else {
82:         let max_depth = num_blocks.ceil_ilog2();
83:         let mut space = 1;
84:         for _ in 0..max_depth {
85:             let num_block_at_iter = num_blocks - space;
86:             let iter_latency = compute_latency_of_one_layer(num_block_at_iter, num_threads);
87:             parallel_expected_latency += iter_latency;
