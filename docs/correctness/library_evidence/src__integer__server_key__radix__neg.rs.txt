102:     where
103:         T: IntegerRadixCiphertext,
104:     {
105:         //z is used to make sure the negation doesn't fill the padding bit
106:         let mut z;
107:         let mut z_b = 0;
108: 
109:         for block in ctxt.blocks_mut() {
110:             if z_b != 0 {
111:                 self.key.unchecked_scalar_add_assign(block, z_b);
112:             }
113:             z = self.key.unchecked_neg_assign_with_correcting_term(block);
114:             block.degree = Degree::new(z as usize - z_b as usize);
115: 
116:             z_b = (z / self.key.message_modulus.0 as u64) as u8;
117:         }
118:     }
